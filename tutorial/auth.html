Flujo Auth: Login → Registro → Intro

Arquitectura del ciclo de autenticación didáctico local: validación de credenciales con ValidationUtils, persistencia mediante SharedPreferences (Prefs.java), transferencia de parámetros en Intent Extras y manejo reactivo de errores en pantalla.

1. MainActivity — Login & Precarga

private void attemptLogin() {
    String email = etEmail.getText() != null ? etEmail.getText().toString().trim() : "";
    String pass  = etPass.getText() != null ? etPass.getText().toString() : "";

    tilEmail.setError(null);
    tilPass.setError(null);
    boolean isValid = true;

    if (!ValidationUtils.isRequired(email)) {
        tilEmail.setError(getString(R.string.common_err_required));
        isValid = false;
    } else if (!ValidationUtils.isEmail(email)) {
        tilEmail.setError(getString(R.string.common_err_email));
        isValid = false;
    }

    if (!ValidationUtils.isRequired(pass)) {
        tilPass.setError(getString(R.string.common_err_required));
        isValid = false;
    } else if (!ValidationUtils.isPassword(pass)) {
        tilPass.setError(getString(R.string.common_err_pass_short));
        isValid = false;
    }

    if (!isValid) {
        Toast.makeText(this, getString(R.string.common_msg_correct_errors), Toast.LENGTH_SHORT).show();
        return;
    }

    Prefs.saveUser(this, email);
    Intent intent = new Intent(this, RegisterActivity.class);
    intent.putExtra("email", email);
    startActivity(intent);
}
private fun attemptLogin() {
    val email = binding.etEmail.text?.toString()?.trim().orEmpty()
    val pass  = binding.etPass.text?.toString().orEmpty()

    binding.tilEmail.error = null
    binding.tilPass.error = null
    var isValid = true

    if (!ValidationUtils.isRequired(email)) {
        binding.tilEmail.error = getString(R.string.common_err_required)
        isValid = false
    } else if (!ValidationUtils.isEmail(email)) {
        binding.tilEmail.error = getString(R.string.common_err_email)
        isValid = false
    }

    if (!ValidationUtils.isRequired(pass)) {
        binding.tilPass.error = getString(R.string.common_err_required)
        isValid = false
    } else if (!ValidationUtils.isPassword(pass)) {
        binding.tilPass.error = getString(R.string.common_err_pass_short)
        isValid = false
    }

    if (!isValid) {
        Toast.makeText(this, getString(R.string.common_msg_correct_errors), Toast.LENGTH_SHORT).show()
        return
    }

    Prefs.saveUser(this, email)

}

3. IntroActivity — Recepción de Extras y Bienvenida

// Recuperación de Extras con fallback a SharedPreferences
String name = getIntent().getStringExtra("name");
String email = getIntent().getStringExtra("email");

if (email == null || email.isEmpty()) {
    email = Prefs.getUser(this);
}
if (name == null || name.isEmpty()) {
    name = email;
}

// Formateo de string i18n con placeholder posicional: ¡Hola, %1$s!
tvGreeting.setText(getString(R.string.common_msg_hello, name));

btnGoHome.setOnClickListener(v -> {
    Intent intent = new Intent(this, HomeActivity.class);
    startActivity(intent);
    finish(); // Cierra el flujo de bienvenida
});
