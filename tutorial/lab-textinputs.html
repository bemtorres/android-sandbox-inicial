Aprende el uso avanzado de inputType en teclados virtuales, maquetación del layout en XML con TextInputLayout, feedback visual con setError y reactividad con TextWatcher.

1. Configuración de los 6 Campos de Entrada

Campo (ID / Hint) inputType Comportamiento Teclado Regla de Validación
tilPlain · Nombre text Texto con autocorrección y mayúscula inicial isRequired (No vacío)
tilEmail · Correo textEmailAddress Teclado con teclas @ y .com directas isEmail (Patterns.EMAIL_ADDRESS)
tilPhone · Teléfono phone Teclado numérico de marcación telefónica isPhone (9 a 15 dígitos)
tilPostal · Código Postal number Teclado numérico puro (0-9) isPostalCode (5 dígitos)
tilPass · Contraseña textPassword Oculta caracteres + botón de toggle de ojo isPassword (Longitud ≥ 6)
tilPin · PIN Numérico numberPassword Numérico con caracteres protegidos isNumericPassword (4 a 6 dígitos)

package com.example.sandbox.sandbox;

import android.os.Bundle;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import com.example.sandbox.R;
import com.example.sandbox.utils.LoggerUtils;
import com.example.sandbox.utils.ValidationUtils;
import com.google.android.material.textfield.TextInputEditText;
import com.google.android.material.textfield.TextInputLayout;

/**
 * TextInputsActivity — Controlador completo para el laboratorio de TextInputs.
 * Demuestra:
 *  1. Captura de 6 tipos de inputs (texto, email, teléfono, numérico, password, PIN).
 *  2. Patrón de validación acumulativa por campo con TextInputLayout.setError().
 *  3. Limpieza de estado y feedback reactivo con TextView y Toast.
 */
public class TextInputsActivity extends AppCompatActivity {

    private static final String TAG = "SANDBOX_INPUTS";

    // Contenedores TextInputLayout para manejo de errores visuales y hints
    private TextInputLayout tilPlain, tilEmail, tilPhone, tilPostal, tilPass, tilPin;

    // Campos de edición de texto
    private TextInputEditText etPlain, etEmail, etPhone, etPostal, etPass, etPin;

    // Vista de preview del estado de validación
    private TextView tvPreview;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_text_inputs);

        // 1. Vinculación de vistas del layout
        bindViews();

        // 2. Configuración de eventos de los botones
        findViewById(R.id.btnValidate).setOnClickListener(v -> validateForm());
        findViewById(R.id.btnClear).setOnClickListener(v -> clearForm());

        LoggerUtils.d(TAG, "TextInputsActivity inicializada correctamente");
    }

    /**
     * Vincula todos los componentes gráficos definidos en activity_text_inputs.xml
     */
    private void bindViews() {
        tilPlain  = findViewById(R.id.tilPlain);
        tilEmail  = findViewById(R.id.tilEmail);
        tilPhone  = findViewById(R.id.tilPhone);
        tilPostal = findViewById(R.id.tilPostal);
        tilPass   = findViewById(R.id.tilPass);
        tilPin    = findViewById(R.id.tilPin);

        etPlain   = findViewById(R.id.etPlain);
        etEmail   = findViewById(R.id.etEmail);
        etPhone   = findViewById(R.id.etPhone);
        etPostal  = findViewById(R.id.etPostal);
        etPass    = findViewById(R.id.etPass);
        etPin     = findViewById(R.id.etPin);

        tvPreview = findViewById(R.id.tvPreview);
    }

    /**
     * Ejecuta la validación acumulativa de todos los campos.
     * Si algún campo es inválido, muestra su mensaje de error específico.
     */
    private void validateForm() {
        String plain  = getTrimmedText(etPlain);
        String email  = getTrimmedText(etEmail);
        String phone  = getTrimmedText(etPhone);
        String postal = getTrimmedText(etPostal);
        String pass   = getTrimmedText(etPass);
        String pin    = getTrimmedText(etPin);

        boolean isValid = true;

        // 1. Validar Nombre (Campo requerido)
        if (!ValidationUtils.isRequired(plain)) {
            tilPlain.setError(getString(R.string.common_err_required));
            isValid = false;
        } else {
            tilPlain.setError(null);
        }

        // 2. Validar Correo Electrónico (Requerido + Formato Email)
        if (!ValidationUtils.isRequired(email)) {
            tilEmail.setError(getString(R.string.common_err_required));
            isValid = false;
        } else if (!ValidationUtils.isEmail(email)) {
            tilEmail.setError(getString(R.string.common_err_email));
            isValid = false;
        } else {
            tilEmail.setError(null);
        }

        // 3. Validar Teléfono (Requerido + Formato Telefónico)
        if (!ValidationUtils.isRequired(phone)) {
            tilPhone.setError(getString(R.string.common_err_required));
            isValid = false;
        } else if (!ValidationUtils.isPhone(phone)) {
            tilPhone.setError(getString(R.string.common_err_phone));
            isValid = false;
        } else {
            tilPhone.setError(null);
        }

        // 4. Validar Código Postal (Requerido + Numérico 5 dígitos)
        if (!ValidationUtils.isRequired(postal)) {
            tilPostal.setError(getString(R.string.common_err_required));
            isValid = false;
        } else if (!ValidationUtils.isPostalCode(postal)) {
            tilPostal.setError(getString(R.string.common_err_postal));
            isValid = false;
        } else {
            tilPostal.setError(null);
        }

        // 5. Validar Contraseña (Requerido + Mínimo 6 caracteres)
        if (!ValidationUtils.isRequired(pass)) {
            tilPass.setError(getString(R.string.common_err_required));
            isValid = false;
        } else if (!ValidationUtils.isPassword(pass)) {
            tilPass.setError(getString(R.string.common_err_pass_short));
            isValid = false;
        } else {
            tilPass.setError(null);
        }

        // 6. Validar PIN (Requerido + 4 a 6 dígitos numéricos)
        if (!ValidationUtils.isRequired(pin)) {
            tilPin.setError(getString(R.string.common_err_required));
            isValid = false;
        } else if (!ValidationUtils.isNumericPassword(pin)) {
            tilPin.setError(getString(R.string.common_err_pass_numeric));
            isValid = false;
        } else {
            tilPin.setError(null);
        }

        // Si existen errores, se muestra notificación y se cancela el flujo
        if (!isValid) {
            Toast.makeText(this, getString(R.string.common_msg_correct_errors), Toast.LENGTH_SHORT).show();
            LoggerUtils.e(TAG, "Validación fallida: uno o más campos contienen errores");
            tvPreview.setText(getString(R.string.actinp_preview_error));
            return;
        }

        // Formulario completamente válido
        String successMsg = getString(R.string.actinp_preview_ok, plain, email, phone);
        tvPreview.setText(successMsg);
        Toast.makeText(this, getString(R.string.common_toast_validation_ok), Toast.LENGTH_SHORT).show();
        LoggerUtils.i(TAG, "Formulario válido: " + successMsg);
    }

    /**
     * Resetea todos los campos de entrada, limpia los errores y restaura el preview.
     */
    private void clearForm() {
        etPlain.setText("");
        etEmail.setText("");
        etPhone.setText("");
        etPostal.setText("");
        etPass.setText("");
        etPin.setText("");

        tilPlain.setError(null);
        tilEmail.setError(null);
        tilPhone.setError(null);
        tilPostal.setError(null);
        tilPass.setError(null);
        tilPin.setError(null);

        tvPreview.setText(getString(R.string.actinp_preview));
        LoggerUtils.d(TAG, "Formulario reseteado exitosamente");
    }

    /**
     * Helper para extraer y recortar el texto de un TextInputEditText de forma segura.
     */
    private String getTrimmedText(TextInputEditText editText) {
        return (editText != null && editText.getText() != null)
                ? editText.getText().toString().trim()
                : "";
    }
}
package com.example.sandbox.sandbox

import android.os.Bundle
import android.util.Patterns
import android.widget.Toast
import androidx.annotation.StringRes
import androidx.appcompat.app.AppCompatActivity
import androidx.core.widget.doAfterTextChanged
import com.example.sandbox.R
import com.example.sandbox.databinding.ActivityTextInputsBinding
import com.example.sandbox.utils.LoggerUtils
import com.google.android.material.textfield.TextInputLayout

/**
 * TextInputsActivity — Implementación idiomática completa en Kotlin + ViewBinding.
 * Incluye gestión de ciclo de vida, validación reactiva en tiempo real y helpers de extensión.
 */
class TextInputsActivity : AppCompatActivity() {

    private val binding by lazy { ActivityTextInputsBinding.inflate(layoutInflater) }
    private val TAG = "SANDBOX_INPUTS"

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(binding.root)

        setupListeners()
        LoggerUtils.d(TAG, "TextInputsActivity (Kotlin) inicializada")
    }

    private fun setupListeners() = with(binding) {
        btnValidate.setOnClickListener { validateForm() }
        btnClear.setOnClickListener { clearForm() }

        // Validación reactiva opcional al escribir
        etEmail.doAfterTextChanged { text ->
            if (text?.toString().isValidEmail()) tilEmail.error = null
        }
    }

    private fun validateForm() = with(binding) {
        val plain  = etPlain.text?.toString().orEmpty().trim()
        val email  = etEmail.text?.toString().orEmpty().trim()
        val phone  = etPhone.text?.toString().orEmpty().trim()
        val postal = etPostal.text?.toString().orEmpty().trim()
        val pass   = etPass.text?.toString().orEmpty().trim()
        val pin    = etPin.text?.toString().orEmpty().trim()

        // Validación fluida por campo utilizando la función de extensión validate()
        val isPlainValid  = tilPlain.validate(plain.isNotBlank(), R.string.common_err_required)
        val isEmailValid  = tilEmail.validate(email.isValidEmail(), R.string.common_err_email)
        val isPhoneValid  = tilPhone.validate(phone.isValidPhone(), R.string.common_err_phone)
        val isPostalValid = tilPostal.validate(postal.isValidPostalCode(), R.string.common_err_postal)
        val isPassValid   = tilPass.validate(pass.length >= 6, R.string.common_err_pass_short)
        val isPinValid    = tilPin.validate(pin.isValidPin(), R.string.common_err_pass_numeric)

        val isFormValid = isPlainValid && isEmailValid && isPhoneValid && 
                          isPostalValid && isPassValid && isPinValid

        if (!isFormValid) {
            Toast.makeText(this@TextInputsActivity, getString(R.string.common_msg_correct_errors), Toast.LENGTH_SHORT).show()
            tvPreview.setText(R.string.actinp_preview_error)
            LoggerUtils.e(TAG, "Validación fallida: campos con errores")
            return@with
        }

        // Formulario válido: actualizar vista previa y notificar al usuario
        tvPreview.text = getString(R.string.actinp_preview_ok, plain, email, phone)
        Toast.makeText(this@TextInputsActivity, getString(R.string.common_toast_validation_ok), Toast.LENGTH_SHORT).show()
        LoggerUtils.i(TAG, "Validación exitosa")
    }

    private fun clearForm() = with(binding) {
        listOf(tilPlain, tilEmail, tilPhone, tilPostal, tilPass, tilPin).forEach { til ->
            til.editText?.text?.clear()
            til.error = null
        }
        tvPreview.setText(R.string.actinp_preview)
        LoggerUtils.d(TAG, "Formulario reseteado exitosamente")
    }

    // Helper de extensión para validar TextInputLayout de forma concisa y declarativa
    private fun TextInputLayout.validate(condition: Boolean, @StringRes errorResId: Int): Boolean {
        return if (condition) {
            error = null
            true
        } else {
            error = getString(errorResId)
            false
        }
    }

    // Helpers de extensión para validación de Strings
    private fun String?.isValidEmail(): Boolean = !this.isNullOrBlank() && Patterns.EMAIL_ADDRESS.matcher(this).matches()
    private fun String?.isValidPhone(): Boolean = !this.isNullOrBlank() && this.matches(Regex("^[+]?[0-9]{9,15}$"))
    private fun String?.isValidPostalCode(): Boolean = !this.isNullOrBlank() && this.matches(Regex("^[0-9]{5}$"))
    private fun String?.isValidPin(): Boolean = !this.isNullOrBlank() && this.matches(Regex("^[0-9]{4,6}$"))
}

  1. Crear el layout res/layout/activity_text_inputs.xml con ScrollView > LinearLayout para scroll fluido.
  2. Añadir los 6 TextInputLayout (tilPlain…tilPin) con sus respectivos TextInputEditText y textos de ayuda.
  3. Declarar TextView id=tvPreview con fondo estilizado para previsualizar el estado del formulario.
  4. En TextInputsActivity.java vincular vistas y listeners para Validar y Limpiar.
  5. Implementar el método validate() con el patrón fail acumulativo llamando a ValidationUtils.
