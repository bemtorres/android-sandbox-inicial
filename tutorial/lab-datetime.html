package com.example.sandbox.sandbox;

import android.app.DatePickerDialog;
import android.app.TimePickerDialog;
import android.os.Bundle;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import com.example.sandbox.R;
import com.example.sandbox.utils.LoggerUtils;
import com.example.sandbox.utils.ValidationUtils;
import java.util.Calendar;
import java.util.Locale;

/**
 * DateTimeActivity — Controlador completo para el laboratorio de Date & Time Pickers.
 * Demuestra:
 *  1. Lanzamiento de DatePickerDialog modal del sistema con Calendar (año, mes base-0, día).
 *  2. Lanzamiento de TimePickerDialog modal del sistema en formato 24h.
 *  3. Formateo de fechas y horas mediante String.format y Locale estándar.
 *  4. Validación de fecha temporal no futura mediante ValidationUtils.isNotFuture().
 */
public class DateTimeActivity extends AppCompatActivity {

    private static final String TAG = "SANDBOX_DATETIME";

    // Vistas para desplegar las selecciones
    private TextView tvDate, tvTime;

    // Calendario para retener la fecha elegida por el usuario
    private Calendar selectedDate = Calendar.getInstance();
    private boolean dateChosen = false;
    private boolean timeChosen = false;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_date_time);

        // 1. Vinculación de vistas
        tvDate = findViewById(R.id.tvDate);
        tvTime = findViewById(R.id.tvTime);

        // 2. Configuración de listeners de eventos
        findViewById(R.id.btnPickDate).setOnClickListener(v -> showDatePicker());
        findViewById(R.id.btnPickTime).setOnClickListener(v -> showTimePicker());
        findViewById(R.id.btnValidateDate).setOnClickListener(v -> validateDate());

        LoggerUtils.d(TAG, "DateTimeActivity inicializada correctamente");
    }

    /**
     * Muestra el DatePickerDialog del sistema configurado con la fecha actual.
     */
    private void showDatePicker() {
        Calendar c = Calendar.getInstance();
        new DatePickerDialog(
                this,
                (view, year, month, day) -> {
                    // Actualizar el calendario interno (recordar: month es 0-indexed)
                    selectedDate.set(year, month, day);
                    dateChosen = true;

                    // Formatear salida en formato dd/MM/yyyy
                    String formatted = String.format(Locale.getDefault(), "%02d/%02d/%04d", day, month + 1, year);
                    tvDate.setText(getString(R.string.acdati_date_prefix, formatted));
                    
                    LoggerUtils.i(TAG, "Fecha seleccionada: " + formatted);
                    Toast.makeText(this, getString(R.string.acdati_date_prefix, formatted), Toast.LENGTH_SHORT).show();
                },
                c.get(Calendar.YEAR),
                c.get(Calendar.MONTH),
                c.get(Calendar.DAY_OF_MONTH)
        ).show();

        LoggerUtils.d(TAG, "DatePickerDialog desplegado");
    }

    /**
     * Muestra el TimePickerDialog del sistema en formato de 24 horas.
     */
    private void showTimePicker() {
        Calendar c = Calendar.getInstance();
        new TimePickerDialog(
                this,
                (view, hour, minute) -> {
                    timeChosen = true;

                    // Formatear salida en formato HH:mm
                    String formatted = String.format(Locale.getDefault(), "%02d:%02d", hour, minute);
                    tvTime.setText(getString(R.string.acdati_time_prefix, formatted));
                    
                    LoggerUtils.i(TAG, "Hora seleccionada: " + formatted);
                    Toast.makeText(this, getString(R.string.acdati_time_prefix, formatted), Toast.LENGTH_SHORT).show();
                },
                c.get(Calendar.HOUR_OF_DAY),
                c.get(Calendar.MINUTE),
                true // is24HourView
        ).show();

        LoggerUtils.d(TAG, "TimePickerDialog desplegado");
    }

    /**
     * Valida que la fecha haya sido seleccionada y que no se encuentre en el futuro.
     */
    private void validateDate() {
        if (!dateChosen) {
            Toast.makeText(this, getString(R.string.acdati_msg_pick_date_first), Toast.LENGTH_SHORT).show();
            LoggerUtils.w(TAG, "Intento de validación sin fecha seleccionada");
            return;
        }

        // Validación de regla de negocio: la fecha debe ser igual o anterior a hoy
        boolean notFuture = ValidationUtils.isNotFuture(selectedDate.getTimeInMillis());
        if (notFuture) {
            Toast.makeText(this, getString(R.string.acdati_msg_date_ok), Toast.LENGTH_SHORT).show();
            LoggerUtils.i(TAG, "Fecha validada exitosamente (no futura)");
        } else {
            tvDate.setError(getString(R.string.common_err_date_future));
            Toast.makeText(this, getString(R.string.common_err_date_future), Toast.LENGTH_SHORT).show();
            LoggerUtils.e(TAG, "Fecha rechazada: se encuentra en el futuro");
        }

        if (!timeChosen) {
            LoggerUtils.w(TAG, "Aviso: Hora aún no elegida");
        }
    }
}
package com.example.sandbox.sandbox

import android.app.DatePickerDialog
import android.app.TimePickerDialog
import android.os.Bundle
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.example.sandbox.R
import com.example.sandbox.databinding.ActivityDateTimeBinding
import com.example.sandbox.utils.LoggerUtils
import com.example.sandbox.utils.ValidationUtils
import java.util.Calendar
import java.util.Locale

/**
 * DateTimeActivity — Implementación idiomática completa en Kotlin + ViewBinding.
 * Demuestra modales DatePickerDialog y TimePickerDialog, binding y validación temporal.
 */
class DateTimeActivity : AppCompatActivity() {

    private val binding by lazy { ActivityDateTimeBinding.inflate(layoutInflater) }
    private val TAG = "SANDBOX_DATETIME"

    private val selectedDate: Calendar = Calendar.getInstance()
    private var dateChosen = false
    private var timeChosen = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(binding.root)

        setupListeners()
        LoggerUtils.d(TAG, "DateTimeActivity (Kotlin) inicializada")
    }

    private fun setupListeners() = with(binding) {
        btnPickDate.setOnClickListener { showDatePicker() }
        btnPickTime.setOnClickListener { showTimePicker() }
        btnValidateDate.setOnClickListener { validateDate() }
    }

    /**
     * Abre el selector DatePickerDialog modal y actualiza el TextView con la fecha formateada.
     */
    private fun showDatePicker() {
        val now = Calendar.getInstance()
        DatePickerDialog(
            this,
            { _, year, month, day ->
                selectedDate.set(year, month, day)
                dateChosen = true

                val formatted = String.format(Locale.getDefault(), "%02d/%02d/%04d", day, month + 1, year)
                binding.tvDate.text = getString(R.string.acdati_date_prefix, formatted)

                LoggerUtils.i(TAG, "Fecha elegida: $formatted")
                Toast.makeText(this, getString(R.string.acdati_date_prefix, formatted), Toast.LENGTH_SHORT).show()
            },
            now.get(Calendar.YEAR),
            now.get(Calendar.MONTH),
            now.get(Calendar.DAY_OF_MONTH)
        ).show()

        LoggerUtils.d(TAG, "DatePickerDialog mostrado")
    }

    /**
     * Abre el selector TimePickerDialog modal en formato 24 horas.
     */
    private fun showTimePicker() {
        val now = Calendar.getInstance()
        TimePickerDialog(
            this,
            { _, hour, minute ->
                timeChosen = true

                val formatted = String.format(Locale.getDefault(), "%02d:%02d", hour, minute)
                binding.tvTime.text = getString(R.string.acdati_time_prefix, formatted)

                LoggerUtils.i(TAG, "Hora elegida: $formatted")
                Toast.makeText(this, getString(R.string.acdati_time_prefix, formatted), Toast.LENGTH_SHORT).show()
            },
            now.get(Calendar.HOUR_OF_DAY),
            now.get(Calendar.MINUTE),
            true // is24HourView
        ).show()

        LoggerUtils.d(TAG, "TimePickerDialog mostrado")
    }

    /**
     * Valida si se eligió fecha y si no es futura mediante ValidationUtils.
     */
    private fun validateDate() {
        if (!dateChosen) {
            Toast.makeText(this, getString(R.string.acdati_msg_pick_date_first), Toast.LENGTH_SHORT).show()
            LoggerUtils.w(TAG, "Validación omitida: falta seleccionar fecha")
            return
        }

        val isNotFuture = ValidationUtils.isNotFuture(selectedDate.timeInMillis)
        if (isNotFuture) {
            Toast.makeText(this, getString(R.string.acdati_msg_date_ok), Toast.LENGTH_SHORT).show()
            LoggerUtils.i(TAG, "Fecha validada correctamente (no futura)")
        } else {
            binding.tvDate.error = getString(R.string.common_err_date_future)
            Toast.makeText(this, getString(R.string.common_err_date_future), Toast.LENGTH_SHORT).show()
            LoggerUtils.e(TAG, "Fecha futura rechazada")
        }

        if (!timeChosen) {
            LoggerUtils.w(TAG, "Aviso: Hora aún no configurada")
        }
    }
}

4. Guía de Desarrollo Paso a Paso

  1. Crear el layout res/layout/activity_date_time.xml con tvDate, tvTime y botones de acción.
  2. En la actividad, instanciar un objeto Calendar.getInstance() para guardar el estado de la fecha.
  3. Implementar DatePickerDialog recordando que los meses en la API clásica de Calendar son 0-indexados (month + 1).
  4. Implementar TimePickerDialog con el flag is24HourView = true.
  5. Al pulsar validar, verificar que selectedDate.getTimeInMillis() <= System.currentTimeMillis() usando ValidationUtils.isNotFuture.
