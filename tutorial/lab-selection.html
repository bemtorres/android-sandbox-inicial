SelectionActivity.java 
btnSubmit.setOnClickListener(v -> {
    // 1. Validar CheckBox obligatorio (Términos y condiciones)
    if (!cbTerms.isChecked()) {
        Toast.makeText(this, getString(R.string.acsele_msg_accept_terms), Toast.LENGTH_SHORT).show();
        return;
    }

    // 2. Obtener RadioButton seleccionado
    int selectedRadioId = radioGroupPlan.getCheckedRadioButtonId();
    String plan = getString(R.string.acsele_plan_free);
    if (selectedRadioId == R.id.rbPro) {
        plan = getString(R.string.acsele_plan_pro);
    } else if (selectedRadioId == R.id.rbEnterprise) {
        plan = getString(R.string.acsele_plan_enterprise);
    }

    // 3. Evaluar Switches
    boolean notifs = switchNotifications.isChecked();
    boolean dark = switchDarkMode.isChecked();

    // 4. Mostrar resumen en TextView
    String summary = getString(R.string.acsele_summary_result, plan, notifs ? "SI" : "NO", dark ? "SI" : "NO");
    tvResult.setText(summary);
    Toast.makeText(this, getString(R.string.acsele_toast_saved), Toast.LENGTH_SHORT).show();
});
